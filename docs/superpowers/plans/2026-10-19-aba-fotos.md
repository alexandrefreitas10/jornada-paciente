# Aba Fotos — Plano de Implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Uma aba `/fotos` que guarda cada antes-e-depois montado (hoje a imagem só é baixada e se perde), e que também monta antes-e-depois de pessoas que não têm card de paciente.

**Architecture:** O cálculo do layout da montagem sai dos componentes para uma função pura (`src/lib/montagem.ts`), travada por teste contra a geometria de hoje, e o desenho no canvas fica num módulo de navegador que os três pontos de montagem compartilham. Três tabelas novas guardam as pastas de quem não tem card, as fotos soltas dessas pastas e as montagens (PNG + miniatura). `patient_files` não é tocada.

**Tech Stack:** Next.js 16 (App Router), React 19, TypeScript, Tailwind v4, Postgres via postgres.js, S3 (`src/lib/s3.ts`), Jest + ts-jest.

**Spec:** `docs/superpowers/specs/2026-10-19-aba-fotos-design.md` — ler as seções citadas em cada task.

```ts
// Layout da montagem antes-e-depois. Função PURA: nada de canvas, nada de DOM.
// O desenho mora em `montagem-canvas.ts` e só executa o que esta função decidiu.
//
// Existe por uma razão: a imagem que sai hoje do card tem de continuar saindo
// IGUAL. As constantes abaixo são cópia literal do que estava embutido em
// `FilesTab.tsx` e `EsteticaTab.tsx`, e `__tests__/lib/montagem.test.ts` fixa a
// geometria que elas produzem — se um pixel mudar de lugar, o teste quebra.
//
// As duas identidades são DIFERENTES de propósito: Fotos é a marca da dra. com
// logo grande, Estética é a do Instituto com logo pequena. Unificar mudaria a
// imagem que o dono manda para o paciente.

export type Identidade = 'fotos' | 'estetica'

export const IDENTIDADES: Identidade[] = ['fotos', 'estetica']

export function ehIdentidade(valor: unknown): valor is Identidade {
  return typeof valor === 'string' && (IDENTIDADES as string[]).includes(valor)
}

/** O caminho da logo de cada identidade. Existe para quem precisa carregar a
 *  logo ANTES de calcular o layout (o layout precisa das medidas dela). */
export function logoDaIdentidade(identidade: Identidade): string {
  return ESTILOS[identidade].logoSrc
}

/** Recorte em coordenadas relativas (0–1) sobre a imagem original. */
export interface Recorte { x: number; y: number; w: number; h: number }
export const RECORTE_CHEIO: Recorte = { x: 0, y: 0, w: 1, h: 1 }

export interface Medidas { largura: number; altura: number }

/** Um `drawImage` pronto: recorte na origem (s*) e destino no canvas (d*). */
export interface Desenho {
  sx: number; sy: number; sw: number; sh: number
  dx: number; dy: number; dw: number; dh: number
}

/** Texto centralizado em `x` (o canvas usa textAlign 'center'). */
export interface Texto { texto: string; x: number; y: number; fonte: string; cor: string }

export interface Layout {
  largura: number
  altura: number
  fundo: string
  logoSrc: string
  logo: { x: number; y: number; w: number; h: number }
  /** [antes, depois] */
  fotos: [Desenho, Desenho]
  /** ANTES, DEPOIS, data do antes, data do depois — nesta ordem. */
  textos: Texto[]
}

/** dd/mm/aaaa, hh:mm no fuso de quem abriu a tela — o que `FilesTab` imprime. */
function dataComHora(iso: string): string {
  return new Date(iso).toLocaleString('pt-BR', {
    day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit',
  })
}

/** dd/mm/aaaa fatiado da string ISO — o que `EsteticaTab` imprime. Fatiar a ISO
 *  lê a data em UTC, então foto enviada à noite em Brasília aparece com o dia
 *  seguinte. É defeito conhecido (spec §8) e fica como está de propósito:
 *  consertar aqui mudaria a imagem que o dono já manda hoje. */
function dataSemHora(iso: string): string {
  if (!iso) return ''
  const [y, m, d] = iso.slice(0, 10).split('-')
  return `${d}/${m}/${y}`
}

interface Estilo {
  gap: number
  labelH: number
  padding: number
  logoH: number
  logoGap: number
  logoSrc: string
  rotuloFonte: string
  rotuloCor: string
  /** Distância do topo do conteúdo até a linha de base do rótulo. */
  rotuloDy: number
  dataFonte: string
  dataCor: string
  /** Distância do fim da foto até a linha de base da data. */
  dataDy: number
  /** Sobra embaixo, depois da data. */
  alturaExtra: number
  formatarData: (iso: string) => string
}

const ESTILOS: Record<Identidade, Estilo> = {
  fotos: {
    gap: 20, labelH: 80, padding: 40, logoH: 400, logoGap: 16,
    logoSrc: '/logo-dra.png',
    rotuloFonte: 'bold 48px sans-serif', rotuloCor: '#374151', rotuloDy: 56,
    dataFonte: 'bold 36px sans-serif', dataCor: '#374151', dataDy: 50,
    alturaExtra: 24,
    formatarData: dataComHora,
  },
  estetica: {
    gap: 16, labelH: 60, padding: 24, logoH: 80, logoGap: 12,
    logoSrc: '/logo-it.png',
    rotuloFonte: 'bold 40px sans-serif', rotuloCor: '#374151', rotuloDy: 44,
    dataFonte: '28px sans-serif', dataCor: '#6b7280', dataDy: 40,
    alturaExtra: 50,
    formatarData: dataSemHora,
  },
}

export interface FotoDaMontagem {
  medidas: Medidas
  recorte: Recorte
  /** ISO do `created_at` da foto de origem. */
  data: string
}

export function calcularLayout(entrada: {
  identidade: Identidade
  antes: FotoDaMontagem
  depois: FotoDaMontagem
  logo: Medidas
}): Layout {
  const e = ESTILOS[entrada.identidade]
  const a = entrada.antes
  const b = entrada.depois

  // Dimensões efetivas depois do recorte, em pixels da imagem original.
  const efA = { w: a.medidas.largura * a.recorte.w, h: a.medidas.altura * a.recorte.h }
  const efB = { w: b.medidas.largura * b.recorte.w, h: b.medidas.altura * b.recorte.h }

  // As duas saem com a MESMA altura — é isso que torna antes e depois
  // comparáveis. Cada uma mantém a proporção, então a largura é que varia.
  const maxH = Math.max(efA.h, efB.h)
  const wA = efA.w * (maxH / efA.h)
  const wB = efB.w * (maxH / efB.h)

  const largura = wA + wB + e.gap + e.padding * 2
  const altura = e.padding * 2 + e.logoH + e.logoGap + e.labelH + maxH + e.alturaExtra
  const logoW = entrada.logo.largura * (e.logoH / entrada.logo.altura)
  const topoConteudo = e.padding + e.logoH + e.logoGap
  const topoFotos = topoConteudo + e.labelH
  const xA = e.padding + wA / 2
  const xB = e.padding + wA + e.gap + wB / 2
  const yData = topoFotos + maxH + e.dataDy

  return {
    largura,
    altura,
    fundo: '#ffffff',
    logoSrc: e.logoSrc,
    logo: { x: (largura - logoW) / 2, y: e.padding, w: logoW, h: e.logoH },
    fotos: [
      {
        sx: a.recorte.x * a.medidas.largura, sy: a.recorte.y * a.medidas.altura,
        sw: efA.w, sh: efA.h,
        dx: e.padding, dy: topoFotos, dw: wA, dh: maxH,
      },
      {
        sx: b.recorte.x * b.medidas.largura, sy: b.recorte.y * b.medidas.altura,
        sw: efB.w, sh: efB.h,
        dx: e.padding + wA + e.gap, dy: topoFotos, dw: wB, dh: maxH,
      },
    ],
    textos: [
      { texto: 'ANTES', x: xA, y: topoConteudo + e.rotuloDy, fonte: e.rotuloFonte, cor: e.rotuloCor },
      { texto: 'DEPOIS', x: xB, y: topoConteudo + e.rotuloDy, fonte: e.rotuloFonte, cor: e.rotuloCor },
      { texto: e.formatarData(a.data), x: xA, y: yData, fonte: e.dataFonte, cor: e.dataCor },
      { texto: e.formatarData(b.data), x: xB, y: yData, fonte: e.dataFonte, cor: e.dataCor },
    ],
  }
}
```

```ts
import { chaveNome } from './implantes'

// As duas regras da aba Fotos que não dependem de banco nem de navegador.

/**
 * A chave que diz se duas pastas são a mesma pessoa. Normaliza acento, caixa e
 * espaço repetido — nome 100% idêntico cai na mesma pasta, que é a regra que o
 * dono pediu desde o dia em que um card de paciente duplicou numa renovação.
 *
 * Reusa `chaveNome` de `implantes.ts` de propósito: é a mesma pergunta ("é a
 * mesma pessoa?") e duas normalizações separadas divergiriam no dia em que uma
 * delas mudasse.
 *
 * Nome parecido mas não idêntico é pasta diferente, também de propósito:
 * adivinhar semelhança é como o card duplicou.
 */
export function chaveDaPasta(nome: string): string {
  const chave = chaveNome(nome ?? '')
  if (!chave) throw new Error('Nome da pessoa é obrigatório')
  return chave
}

export type Origem =
  | { patientId: number; pessoaId: null }
  | { patientId: null; pessoaId: number }

function idOuNulo(valor: unknown): number | null {
  if (valor == null || valor === '') return null
  const n = Number(valor)
  if (!Number.isInteger(n) || n <= 0) throw new Error('Id inválido')
  return n
}

/**
 * Uma montagem é de um paciente do sistema OU de uma pasta sem card, nunca dos
 * dois e nunca de nenhum. O banco também tem esse CHECK; a função existe para a
 * rota recusar com uma mensagem em português antes de subir arquivo para o S3.
 */
export function origemDaMontagem(entrada: { patientId?: unknown; pessoaId?: unknown }): Origem {
  const paciente = idOuNulo(entrada.patientId)
  const pessoa = idOuNulo(entrada.pessoaId)
  if ((paciente == null) === (pessoa == null)) {
    throw new Error('A montagem tem de ser de um paciente OU de uma pessoa sem card')
  }
  return paciente != null
    ? { patientId: paciente, pessoaId: null }
    : { patientId: null, pessoaId: pessoa as number }
}
```

```ts
// Acesso ao banco da aba Fotos. Três tabelas: as pastas de quem não tem card,
// as fotos soltas dessas pastas, e as montagens já prontas.

export interface FotoPessoa {
  id: number
  nome: string
  chave_nome: string
  created_at: string
  created_by: string | null
}

export interface PastaComContagem extends FotoPessoa {
  fotos: number
}

export interface FotoAvulsa {
  id: number
  pessoa_id: number
  s3_key: string
  original_name: string
  created_at: string
  created_by: string | null
}

export interface Montagem {
  id: number
  patient_id: number | null
  pessoa_id: number | null
  nome: string
  identidade: Identidade
  s3_key: string
  s3_key_thumb: string
  data_antes: string | null
  data_depois: string | null
  created_at: string
  created_by: string | null
}

/**
 * Acha-ou-cria a pasta pela chave normalizada. O `ON CONFLICT` é o que garante
 * a regra sob corrida: dois envios do mesmo nome ao mesmo tempo não podem virar
 * duas pastas. O `DO UPDATE` é um no-op proposital — serve só para o RETURNING
 * devolver a linha existente, e preserva o nome como foi digitado na primeira
 * vez (a pasta não é renomeável nesta versão).
 */
export async function acharOuCriarPessoa(nome: string, createdBy?: string | null): Promise<FotoPessoa> {
  await initSchema()
  const chave = chaveDaPasta(nome)
  const [row] = await sql<FotoPessoa[]>`
    INSERT INTO foto_pessoas (nome, chave_nome, created_by)
    VALUES (${nome.trim()}, ${chave}, ${createdBy ?? null})
    ON CONFLICT (chave_nome) DO UPDATE SET nome = foto_pessoas.nome
    RETURNING *
  `
  return row
}

export async function listarPessoas(): Promise<PastaComContagem[]> {
  await initSchema()
  return sql<PastaComContagem[]>`
    SELECT p.*, COUNT(f.id)::int AS fotos
    FROM foto_pessoas p
    LEFT JOIN fotos_avulsas f ON f.pessoa_id = p.id AND f.deleted_at IS NULL
    GROUP BY p.id
    ORDER BY p.nome
  `
}

export async function criarFotoAvulsa(entrada: {
  pessoaId: number
  s3Key: string
  originalName: string
  createdBy?: string | null
}): Promise<FotoAvulsa> {
  await initSchema()
  const [row] = await sql<FotoAvulsa[]>`
    INSERT INTO fotos_avulsas (pessoa_id, s3_key, original_name, created_by)
    VALUES (${entrada.pessoaId}, ${entrada.s3Key}, ${entrada.originalName}, ${entrada.createdBy ?? null})
    RETURNING *
  `
  return row
}

export async function listarFotosAvulsas(pessoaId: number): Promise<FotoAvulsa[]> {
  await initSchema()
  return sql<FotoAvulsa[]>`
    SELECT * FROM fotos_avulsas
    WHERE pessoa_id = ${pessoaId} AND deleted_at IS NULL
    ORDER BY created_at DESC
  `
}

/** Nunca devolve foto excluída: um "excluído" não pode ser servido nem baixado
 *  (mesma regra de `getFileById` em `patient-files.ts`). */
export async function acharFotoAvulsa(id: number): Promise<FotoAvulsa | null> {
  await initSchema()
  const [row] = await sql<FotoAvulsa[]>`
    SELECT * FROM fotos_avulsas WHERE id = ${id} AND deleted_at IS NULL
  `
  return row ?? null
}

export async function softDeleteFotoAvulsa(id: number): Promise<boolean> {
  await initSchema()
  const linhas = await sql`
    UPDATE fotos_avulsas SET deleted_at = NOW() WHERE id = ${id} AND deleted_at IS NULL RETURNING id
  `
  return linhas.length > 0
}

/**
 * Grava a montagem. O `nome` é CÓPIA do nome na hora, resolvida aqui no
 * servidor: a galeria tem de continuar legível se o card do paciente for
 * arquivado, e os componentes do card recebem só o id do paciente, nunca o
 * nome. Pessoa que não existe é erro alto, não um "Sem nome" silencioso.
 */
export async function criarMontagem(entrada: {
  patientId: number | null
  pessoaId: number | null
  identidade: Identidade
  s3Key: string
  s3KeyThumb: string
  dataAntes: string | null
  dataDepois: string | null
  createdBy?: string | null
}): Promise<Montagem> {
  await initSchema()
  const nome = entrada.patientId != null
    ? (await sql<{ name: string }[]>`SELECT name FROM patients WHERE id = ${entrada.patientId}`)[0]?.name
    : (await sql<{ nome: string }[]>`SELECT nome FROM foto_pessoas WHERE id = ${entrada.pessoaId}`)[0]?.nome
  if (!nome) throw new Error('Pessoa da montagem não encontrada')

  const [row] = await sql<Montagem[]>`
    INSERT INTO antes_depois
      (patient_id, pessoa_id, nome, identidade, s3_key, s3_key_thumb, data_antes, data_depois, created_by)
    VALUES (
      ${entrada.patientId}, ${entrada.pessoaId}, ${nome}, ${entrada.identidade},
      ${entrada.s3Key}, ${entrada.s3KeyThumb}, ${entrada.dataAntes}, ${entrada.dataDepois},
      ${entrada.createdBy ?? null}
    )
    RETURNING *
  `
  return row
}

export async function listarMontagens(): Promise<Montagem[]> {
  await initSchema()
  return sql<Montagem[]>`
    SELECT * FROM antes_depois WHERE deleted_at IS NULL ORDER BY created_at DESC
  `
}

export async function acharMontagem(id: number): Promise<Montagem | null> {
  await initSchema()
  const [row] = await sql<Montagem[]>`
    SELECT * FROM antes_depois WHERE id = ${id} AND deleted_at IS NULL
  `
  return row ?? null
}

export async function softDeleteMontagem(id: number): Promise<boolean> {
  await initSchema()
  const linhas = await sql`
    UPDATE antes_depois SET deleted_at = NOW() WHERE id = ${id} AND deleted_at IS NULL RETURNING id
  `
  return linhas.length > 0
}
```

## Task 6: O desenho no canvas, compartilhado

**Files:**
- Create: `src/lib/montagem-canvas.ts`

Este módulo é o único lugar que desenha. Ele recebe as duas imagens já carregadas (quem carrega é o chamador, porque cada tela tem a sua rota de imagem) e devolve o PNG e a miniatura.

- [ ] **Step 1: Escrever `src/lib/montagem-canvas.ts`**

```ts
import { calcularLayout, logoDaIdentidade, type Identidade, type Recorte } from './montagem'

// O desenho da montagem. Vive separado de `montagem.ts` porque depende de
// canvas e de Image: a geometria é testada sem navegador, aqui só se executa o
// que ela decidiu.

/** Largura máxima da miniatura da galeria. A montagem de duas fotos de celular
 *  passa de 4000px de altura e o PNG tem dezenas de MB; uma grade com 30 deles
 *  travaria o navegador. O que o dono baixa continua sendo o PNG inteiro. */
const THUMB_MAX_W = 700

function carregarImagem(src: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    img.onload = () => resolve(img)
    img.onerror = () => reject(new Error(`Não foi possível carregar ${src}`))
    img.src = src
  })
}

function paraBlob(canvas: HTMLCanvasElement, tipo: string, qualidade?: number): Promise<Blob> {
  return new Promise((resolve, reject) => {
    canvas.toBlob(
      blob => (blob ? resolve(blob) : reject(new Error('Canvas não gerou imagem'))),
      tipo,
      qualidade,
    )
  })
}

export interface FotoParaMontar {
  /** Imagem já carregada, servida pela PRÓPRIA origem — imagem de outro domínio
   *  suja o canvas e faz `toBlob` lançar erro de segurança. */
  img: HTMLImageElement
  recorte: Recorte
  /** ISO do `created_at` da foto. */
  data: string
}

export interface MontagemGerada {
  png: Blob
  thumb: Blob
  largura: number
  altura: number
}

export async function gerarMontagem(entrada: {
  identidade: Identidade
  antes: FotoParaMontar
  depois: FotoParaMontar
}): Promise<MontagemGerada> {
  const { identidade, antes, depois } = entrada
  const medidas = (img: HTMLImageElement) => ({ largura: img.naturalWidth, altura: img.naturalHeight })

  // A logo de cada identidade é decidida pela identidade, não pelo chamador — e
  // tem de ser carregada antes do cálculo, porque o layout precisa das medidas
  // dela para centralizá-la.
  const logo = await carregarImagem(logoDaIdentidade(identidade))

  const layout = calcularLayout({
    identidade,
    antes: { medidas: medidas(antes.img), recorte: antes.recorte, data: antes.data },
    depois: { medidas: medidas(depois.img), recorte: depois.recorte, data: depois.data },
    logo: medidas(logo),
  })

  const canvas = document.createElement('canvas')
  canvas.width = layout.largura
  canvas.height = layout.altura
  const ctx = canvas.getContext('2d')!

  ctx.fillStyle = layout.fundo
  ctx.fillRect(0, 0, layout.largura, layout.altura)
  ctx.drawImage(logo, layout.logo.x, layout.logo.y, layout.logo.w, layout.logo.h)

  const fontes: [FotoParaMontar, FotoParaMontar] = [antes, depois]
  layout.fotos.forEach((d, i) => {
    ctx.drawImage(fontes[i].img, d.sx, d.sy, d.sw, d.sh, d.dx, d.dy, d.dw, d.dh)
  })

  ctx.textAlign = 'center'
  for (const t of layout.textos) {
    ctx.fillStyle = t.cor
    ctx.font = t.fonte
    ctx.fillText(t.texto, t.x, t.y)
  }

  const png = await paraBlob(canvas, 'image/png')

  const escala = Math.min(1, THUMB_MAX_W / layout.largura)
  const mini = document.createElement('canvas')
  mini.width = Math.max(1, Math.round(layout.largura * escala))
  mini.height = Math.max(1, Math.round(layout.altura * escala))
  const miniCtx = mini.getContext('2d')!
  miniCtx.fillStyle = layout.fundo
  miniCtx.fillRect(0, 0, mini.width, mini.height)
  miniCtx.drawImage(canvas, 0, 0, mini.width, mini.height)
  const thumb = await paraBlob(mini, 'image/jpeg', 0.8)

  return { png, thumb, largura: layout.largura, altura: layout.altura }
}

/** Dispara o download no navegador. Separado de `gerarMontagem` porque a tela
 *  baixa PRIMEIRO e salva depois: se o POST falhar, o dono já está com a
 *  imagem na mão, que é o que ele precisa no momento. */
export function baixarBlob(blob: Blob, nomeArquivo: string): void {
  const url = URL.createObjectURL(blob)
  const link = document.createElement('a')
  link.href = url
  link.download = nomeArquivo
  link.click()
  URL.revokeObjectURL(url)
}

/** Sobe a montagem para a aba Fotos. Devolve o id gravado; quem chama guarda o
 *  id para o segundo clique não gravar uma segunda cópia. */
export async function salvarMontagem(entrada: {
  png: Blob
  thumb: Blob
  identidade: Identidade
  patientId?: number | null
  pessoaId?: number | null
  dataAntes: string
  dataDepois: string
}): Promise<number> {
  const fd = new FormData()
  fd.append('png', entrada.png, 'montagem.png')
  fd.append('thumb', entrada.thumb, 'montagem-thumb.jpg')
  fd.append('identidade', entrada.identidade)
  if (entrada.patientId != null) fd.append('patientId', String(entrada.patientId))
  if (entrada.pessoaId != null) fd.append('pessoaId', String(entrada.pessoaId))
  fd.append('dataAntes', entrada.dataAntes)
  fd.append('dataDepois', entrada.dataDepois)

  const res = await fetch('/api/fotos/montagens', { method: 'POST', body: fd })
  if (!res.ok) {
    const body = await res.json().catch(() => ({}))
    throw new Error(body.error || 'Não foi possível guardar na aba Fotos')
  }
  const row = await res.json()
  return row.id as number
}
```
